      const height = viewerRef.current.camera.positionCartographic.height;
      viewerRef.current.camera.zoomOut(height * 0.3);
    }
  };

  // RELEASE LOCK FUNCTION - Fly back to original starting coordinates
  const handleReleaseLock = () => {
    if (viewerRef.current) {
      // 1. Break the lock
      viewerRef.current.trackedEntity = undefined;
      
      // 2. Clear selected state safely
      if (selectedSatRef.current && selectedSatRef.current.path) {
          selectedSatRef.current.path.show = false;
      }
      selectedSatRef.current = null;
      setSelectedEntity(null);

      // 3. Cinematic fly back to the original starting altitude and coordinates
      viewerRef.current.camera.flyTo({
        destination: Cesium.Cartesian3.fromDegrees(78.9629, 20.5937, 8000000.0),
        duration: 2.0 // 2 seconds animation duration
      });
    }
  };

  const getFilterClass = () => {
    switch (tacticalMode) {
      case 'NVG': return 'map-filter-nvg';
      case 'FLIR': return 'map-filter-flir';
      case 'NOIR': return 'map-filter-noir';
      default: return 'map-filter-normal';
    }
  };

  return (
    <div className="absolute inset-0 bg-black">
      <div 
        ref={containerRef} 
        className={`w-full h-full transition-all duration-700 ease-in-out ${getFilterClass()}`}
      />

      {/* RELEASE LOCK BUTTON (Appears only when tracking) */}
      {isTracking && (
        <div className="absolute top-24 left-1/2 transform -translate-x-1/2 z-50 pointer-events-auto">
          <button
            onClick={handleReleaseLock}
            className="px-5 py-2 border border-[#00ff9f]/50 bg-black/80 text-[#00ff9f] font-mono text-[12px] tracking-widest hover:bg-[#00ff9f]/20 hover:border-[#00ff9f] shadow-[0_0_15px_rgba(0,255,159,0.2)] transition-all flex items-center gap-3 backdrop-blur-sm"
          >
            <span className="animate-pulse text-red-500">●</span> 
            [ RELEASE TARGET LOCK ]
          </button>
        </div>
      )}

      {/* ON-SCREEN ZOOM CONTROLS */}
      <div className="absolute right-6 bottom-[220px] z-50 flex flex-col gap-2 font-mono text-[16px] pointer-events-auto">
        <button
          onClick={handleZoomIn}
          className="w-10 h-10 border border-[#00ff9f]/40 bg-black/60 text-[#00ff9f] hover:bg-[#00ff9f]/20 hover:border-[#00ff9f] shadow-[0_0_10px_rgba(0,255,159,0.1)] flex items-center justify-center backdrop-blur-sm transition-all"
        >
          +
        </button>
        <button
          onClick={handleZoomOut}
          className="w-10 h-10 border border-[#00ff9f]/40 bg-black/60 text-[#00ff9f] hover:bg-[#00ff9f]/20 hover:border-[#00ff9f] shadow-[0_0_10px_rgba(0,255,159,0.1)] flex items-center justify-center backdrop-blur-sm transition-all"
        >
          -
        </button>
      </div>
      
      {/* HUD DETECTION MODE TOGGLE */}
      <div className="absolute right-6 bottom-24 z-50 flex flex-col gap-2 font-mono text-[11px] tracking-wider pointer-events-auto">
        <div className="text-[#00ff9f] mb-1 opacity-80 border-b border-[#00ff9f]/30 pb-1">HUD TARGETING</div>
        {['OFF', 'SPARSE', 'PANOPTIC'].map((mode) => (
          <button
            key={mode}
            onClick={() => setDetectionMode(mode as any)}
            className={`px-4 py-1.5 border backdrop-blur-sm transition-all text-left w-32 ${
              detectionMode === mode 
                ? 'border-[#00ff9f] bg-[#00ff9f]/10 text-[#00ff9f] shadow-[0_0_10px_rgba(0,255,159,0.2)]' 
                : 'border-[#00ff9f]/20 bg-black/60 text-[#00ff9f]/50 hover:bg-[#00ff9f]/5 hover:border-[#00ff9f]/50'
            }`}
          >
            <span className="inline-block w-4">{detectionMode === mode ? '■' : '□'}</span> {mode}
          </button>
        ))}
      </div>

      <style>{`
        .cesium-viewer-bottom { display: none !important; }
        .cesium-widget-credits { display: none !important; }
      `}</style>
    </div>
  );
}
